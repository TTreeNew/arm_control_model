### 控制器最小骨架代码
```cpp
class MyController : public controller_interface::ControllerInterface
{
public:
  // ① 声明：我要借哪些接口
  InterfaceConfiguration command_interface_configuration() const override;
  InterfaceConfiguration state_interface_configuration() const override;

  // ② 生命周期
  CallbackReturn on_configure(const State &) override;    // 建资源
  CallbackReturn on_activate(const State &) override;     // 初始化

  // ③ 核心循环
  return_type update(const rclcpp::Time & t, const rclcpp::Duration & dt) override
  {
    read_state_from_state_interfaces(state_current_);   // ① 读
    compute_desired(t);                                  // ② 算
    for (i) cmd_[i].get().set_value(desired_[i]);        // ③ 写
    return return_type::OK;
  }

private:
  std::vector<std::reference_wrapper<LoanedCommandInterface>> cmd_;    // 写通道
  std::vector<std::reference_wrapper<LoanedStateInterface>>   state_;  // 读通道
  JointTrajectoryPoint state_current_, state_desired_;
};
```
